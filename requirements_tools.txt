# Specialist-model stack for run_tools.py (see README "From Routing Tokens to Real Outputs").
#
# ONE environment runs both the Olympus router and every specialist. The version
# window below is narrow and was found empirically -- please do not loosen it
# without re-running scripts/smoke_test_tokens.py.
#
#   conda activate olympus
#   pip install torch==2.6.0 torchvision==0.21.0 --index-url https://download.pytorch.org/whl/cu124
#   pip install -r requirements_tools.txt
#   bash scripts/install_specialists.sh
#
# --------------------------------------------------------------------------
# WHY transformers IS PINNED TO EXACTLY 4.49.0
# --------------------------------------------------------------------------
# * transformers 5.x  -> removes CLIPVisionTransformer, so mipha's vision tower
#                        fails to import outright.
# * transformers 4.57 -> imports and generates, but the router silently emits
#                        WRONG text instead of routing tokens. This failure is
#                        silent: no exception, just garbage output. Do not
#                        upgrade without checking predict.py still reproduces the
#                        routing tokens shown in the README.
# * transformers 4.49 -> verified correct for the router AND new enough for
#                        Qwen2.5-VL (Qwen-Image's text encoder) and diffusers
#                        0.39 pipelines.
# --------------------------------------------------------------------------
transformers==4.49.0
tokenizers>=0.21,<0.22

# diffusers 0.39 is required for QwenImagePipeline and WanPipeline (Wan2.2).
diffusers==0.39.0
peft>=0.17.0          # hard requirement of diffusers 0.39
accelerate>=1.4.0
huggingface_hub>=0.29.1
safetensors
sentencepiece
protobuf

numpy<2
pillow
scipy
einops
opencv-python==4.11.0.86
matplotlib

# timm 1.0.x is REQUIRED: transformers>=4.49 ships a timm_wrapper model that
# imports timm.data.ImageNetInfo, which only exists from timm 1.0. Without it,
# GroundingDINO (<image_ground>) and InstructIR (<image_deblur>/<image_denoise>/
# <image_derain>) fail to import.
#
# pip will warn that controlnet_aux 0.0.9 wants timm<=0.6.7. That pin is stale:
# the detectors Olympus uses (OpenPose, DWPose, NormalBAE, PidiNet) were verified
# working on timm 1.0.15. Install timm AFTER controlnet_aux and ignore the warning.
timm==1.0.15

# condition-map estimators for controllable generation (pose/normal/scribble)
controlnet_aux==0.0.9

# video I/O
imageio[ffmpeg]
imageio-ffmpeg

# 3D mesh export / preview
trimesh
pygltflib

# background removal used by the TripoSR image-to-3D path
rembg[cpu]
onnxruntime

# InstructIR config loading
pyyaml
