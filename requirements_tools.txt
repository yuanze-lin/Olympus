# Specialist-model stack for run_tools.py (see README "From Routing Tokens to Real Outputs").
#
# ONE environment runs both the Olympus router and every specialist. The version
# window below is narrow and was found empirically -- please do not loosen it
# without re-running scripts/smoke_test_tokens.py.
#
#   conda activate olympus
#   pip install torch==2.6.0 torchvision==0.21.0 --index-url https://download.pytorch.org/whl/cu124
#   pip install -r requirements_tools.txt
#   bash scripts/install_specialists.sh
#
# --------------------------------------------------------------------------
# WHY transformers IS PINNED TO 4.57.x
# --------------------------------------------------------------------------
# The usable window is transformers >= 4.50, < 5.0. Verified on 4.57.6.
#
# * transformers 5.x   -> removes CLIPVisionTransformer, so mipha's vision tower
#                         fails to import outright.
# * transformers <=4.49 -> too old for Qwen-Image: loading its Qwen2.5-VL text
#                         encoder dies with "'dict' object has no attribute
#                         'to_dict'". The router itself works on 4.49, but the
#                         default <image_gen>/<image_edit> backends do not.
#
# NOTE: on transformers >= 4.50 the router used to fail SILENTLY -- it produced
# fluent text but no routing tokens, because mipha passed no position_ids and
# every token after the first decoded at position 0. That is fixed in
# mipha/model/language_model/mipha_phi.py. If you change this pin, always check
# that predict.py still reproduces the routing tokens shown in the README.
# --------------------------------------------------------------------------
transformers==4.57.6
tokenizers>=0.22,<0.23

# diffusers 0.39 is required for QwenImagePipeline and WanPipeline (Wan2.2).
diffusers==0.39.0
peft>=0.17.0          # hard requirement of diffusers 0.39
accelerate>=1.14.0
huggingface_hub>=0.36.0
safetensors
sentencepiece
protobuf

numpy<2
pillow
scipy
einops
opencv-python==4.11.0.86
matplotlib

# condition-map estimators for controllable generation (pose/normal/scribble)
controlnet_aux==0.0.9

# NOTE: timm is deliberately NOT pinned here. transformers >= 4.49 needs
# timm >= 1.0 (it imports timm.data.ImageNetInfo via its timm_wrapper model),
# but controlnet_aux 0.0.9 declares timm<=0.6.7. Pinning both in this file makes
# pip fail with ResolutionImpossible, so scripts/install_specialists.sh upgrades
# timm to 1.0.15 afterwards. That controlnet_aux pin is stale -- the detectors
# Olympus uses (OpenPose, DWPose, NormalBAE, PidiNet) are verified on 1.0.15.
